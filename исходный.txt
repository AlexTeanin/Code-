class SplashScreen:
    def __init__(self, root, image_path, min_duration=2000):  # ← 2 секунды
        self.root = root
        self.min_duration = min_duration
        self.splash_window = None
        self.image_path = image_path
        self.photo = None
        self.progress = None
        self.start_time = None
        self.splash_width = 0
        self.splash_height = 0
        self.app_ready = False  # ← НОВЫЙ ФЛАГ
        self.create_splash()
    
    def create_splash(self):  # ← 4 пробела отступа!
        self.splash_window = ctk.CTkToplevel(self.root)
        self.splash_window.title("")
        self.splash_window.overrideredirect(True)
        self.splash_window.attributes('-transparentcolor', '#000001')
        self.splash_window.configure(fg_color='#000001')
        
        screen_width = self.splash_window.winfo_screenwidth()
        screen_height = self.splash_window.winfo_screenheight()
        
        try:
            self.original_image = Image.open(self.image_path).convert("RGBA")
            orig_width, orig_height = self.original_image.size
            scale_factor = 1/3
            max_width = min(orig_width, screen_width - 100)
            max_height = min(orig_height, screen_height - 100)
            ratio = min(max_width / orig_width, max_height / orig_height)
            self.splash_width = int(orig_width * ratio * scale_factor)
            self.splash_height = int(orig_height * ratio * scale_factor)
            self.resized_image = self.original_image.resize(
                (self.splash_width, self.splash_height),
                Image.Resampling.LANCZOS
            )
            self.photo = ctk.CTkImage(
                light_image=self.resized_image,
                dark_image=self.resized_image,
                size=(self.splash_width, self.splash_height)
            )
            label = ctk.CTkLabel(
                self.splash_window,
                image=self.photo,
                text="",
                fg_color='#000001'
            )
            label.place(x=0, y=0, relwidth=1.0, relheight=1.0)
        except Exception as e:
            print(f"⚠️ Не удалось загрузить изображение заставки: {e}")
            self.splash_width = 500
            self.splash_height = 300
        
        x = (screen_width - self.splash_width) // 2
        y = (screen_height - self.splash_height) // 2
        self.splash_window.geometry(f"{self.splash_width}x{self.splash_height}+{x}+{y}")
        self.splash_window.attributes('-topmost', True)
        
        self.progress = ctk.CTkProgressBar(
            self.splash_window,
            mode="indeterminate",
            width=self.splash_width - 40,
            height=10,
            fg_color='#000001'
        )
        self.progress.place(x=20, y=self.splash_height - 30)
        self.progress.start()
        self.start_time = time.time()
        self.root.update()
    
    def can_close(self):  # ← 4 пробела отступа!
        """Проверяет, можно ли закрыть заставку"""
        if self.start_time is None:
            return self.app_ready
        elapsed = (time.time() - self.start_time) * 1000
        return elapsed >= self.min_duration and self.app_ready
    
    def set_app_ready(self):  # ← НОВЫЙ МЕТОД!
        """Вызывается когда приложение полностью загружено"""
        self.app_ready = True
    
    def close(self):  # ← 4 пробела отступа!
        while not self.can_close():
            time.sleep(0.1)
            self.root.update()
        if self.progress:
            self.progress.stop()
        if self.splash_window:
            self.splash_window.destroy()
            self.splash_window = None

# Класс для всплывающих подсказок (tooltips)
# ------------------------------------------------------------
class ToolTip:
    def __init__(self, widget, text):
        self.widget = widget
        self.text = text
        self.tip_window = None
        widget.bind('<Enter>', self.show_tip)
        widget.bind('<Leave>', self.hide_tip)
    
    def show_tip(self, event=None):
        if self.tip_window or not self.text:
            return
        x, y, _, _ = self.widget.bbox("insert")
        x += self.widget.winfo_rootx() + 25
        y += self.widget.winfo_rooty() + 25
        self.tip_window = tw = tk.Toplevel(self.widget)
        tw.wm_overrideredirect(True)
        tw.wm_geometry(f"+{x}+{y}")
        label = tk.Label(tw, text=self.text, justify=tk.LEFT,
                        background="#ffffe0", relief=tk.SOLID, borderwidth=1,
                        font=("Segoe UI", "10", "normal"))
        label.pack()
    
    def hide_tip(self, event=None):
        if self.tip_window:
            self.tip_window.destroy()
            self.tip_window = None

# Класс для работы с базой данных SQLite
# ------------------------------------------------------------
class Database:
    def __init__(self, db_path=None):
        self.db_path = db_path
        self.connection = None
        if db_path:
            self.connect(db_path)
    
    def connect(self, db_path):
        try:
            self.connection = sqlite3.connect(db_path)
            self.connection.row_factory = sqlite3.Row
            cursor = self.connection.cursor()
            cursor.execute('''
                CREATE TABLE IF NOT EXISTS metal_data (
                    article TEXT PRIMARY KEY,
                    data TEXT NOT NULL
                )
            ''')
            cursor.execute('''
                CREATE TABLE IF NOT EXISTS roll_articles (
                    article TEXT PRIMARY KEY,
                    area REAL NOT NULL
                )
            ''')
            cursor.execute('''
                CREATE TABLE IF NOT EXISTS f12_production_data (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    date TEXT NOT NULL,
                    part_article TEXT NOT NULL,
                    part_name TEXT,
                    order_num TEXT,
                    good_qty INTEGER DEFAULT 0,
                    defect_qty INTEGER DEFAULT 0,
                    UNIQUE(date, part_article, order_num)
                )
                ''')
            self.connection.commit()
            self.db_path = db_path
            return True
        except Exception as e:
            print(f"Ошибка подключения к БД: {e}")
            self.connection = None
            return False
    

    def save_f12_record(self, date, part_article, part_name, order_num, good_qty, defect_qty):
        """Сохраняет запись из F12 в БД"""
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute('''
            INSERT OR REPLACE INTO f12_production_data 
            (date, part_article, part_name, order_num, good_qty, defect_qty)
            VALUES (?, ?, ?, ?, ?, ?)
        ''', (date, part_article, part_name, order_num, good_qty, defect_qty))
        self.connection.commit()
        return True

    def load_f12_records(self, date_from=None, date_to=None):
        """Загружает записи из F12 за период"""
        if not self.connection:
            return []
        cursor = self.connection.cursor()
        if date_from and date_to:
            cursor.execute('''
                SELECT date, part_article, part_name, order_num, good_qty, defect_qty 
                FROM f12_production_data 
                WHERE date BETWEEN ? AND ?
                ORDER BY date, part_article
            ''', (date_from, date_to))
        else:
            cursor.execute('''
                SELECT date, part_article, part_name, order_num, good_qty, defect_qty 
                FROM f12_production_data 
                ORDER BY date, part_article
            ''')
        rows = cursor.fetchall()
        return [dict(row) for row in rows]

    def delete_f12_records(self, date, part_article=None):
        """Удаляет записи F12 за дату (опционально по артикулу)"""
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        if part_article:
            cursor.execute('''
                DELETE FROM f12_production_data 
                WHERE date = ? AND part_article = ?
            ''', (date, part_article))
        else:
            cursor.execute('''
                DELETE FROM f12_production_data 
                WHERE date = ?
            ''', (date,))
        self.connection.commit()
        return True

    def clear_f12_table(self):
        """Очищает всю таблицу F12"""
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute("DELETE FROM f12_production_data")
        self.connection.commit()
        return True

    def load_all_records(self):
        if not self.connection:
            return {}
        cursor = self.connection.cursor()
        cursor.execute("SELECT article, data FROM metal_data")
        rows = cursor.fetchall()
        records = {}
        for row in rows:
            try:
                records[row['article']] = json.loads(row['data'])
            except:
                pass
        return records
    
    def save_article(self, article, data):
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute(
            "REPLACE INTO metal_data (article, data) VALUES (?, ?)",
            (article, json.dumps(data, ensure_ascii=False))
        )
        self.connection.commit()
        return True
    
    def save_all_records(self, records):
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute("DELETE FROM metal_data")
        for article, data in records.items():
            cursor.execute(
                "INSERT INTO metal_data (article, data) VALUES (?, ?)",
                (article, json.dumps(data, ensure_ascii=False))
            )
        self.connection.commit()
        return True
    
    def delete_article(self, article):
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute("DELETE FROM metal_data WHERE article = ?", (article,))
        self.connection.commit()
        return True
    
    def save_roll_article(self, article, area):
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute(
            "REPLACE INTO roll_articles (article, area) VALUES (?, ?)",
            (article, area)
        )
        self.connection.commit()
        return True
    
    def delete_roll_article(self, article):
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute("DELETE FROM roll_articles WHERE article = ?", (article,))
        self.connection.commit()
        return True
    
    def save_all_roll_articles(self, articles, areas):
        if not self.connection:
            return False
        cursor = self.connection.cursor()
        cursor.execute("DELETE FROM roll_articles")
        for art, area in zip(articles, areas):
            cursor.execute(
                "INSERT INTO roll_articles (article, area) VALUES (?, ?)",
                (art, area)
            )
        self.connection.commit()
        return True
    
    def load_roll_articles(self):
        if not self.connection:
            return [], []
        cursor = self.connection.cursor()
        cursor.execute("SELECT article, area FROM roll_articles")
        rows = cursor.fetchall()
        articles = []
        areas = []
        for row in rows:
            articles.append(row['article'])
            areas.append(row['area'])
        return articles, areas
    
    def close(self):
        if self.connection:
            self.connection.close()
            self.connection = None

def get_cell_coordinate(cell_ref):
    """Извлекает номер строки из адреса ячейки (например, 'A1' -> 1)"""
    coord = []
    for c in cell_ref:
        if c.isdigit():
            coord.append(c)
    return int(''.join(coord)) if coord else 0

def extract_excel_data_raw(file_path):
    """Извлекает данные из .xlsx напрямую через ZIP/XML, игнорируя стили."""
    if not os.path.exists(file_path):
        print(f"❌ Файл не найден: {file_path}")
        return None
    
    print(f"🔧 Прямое чтение XML из файла: {file_path}")
    try:
        with zipfile.ZipFile(file_path, 'r') as zip_ref:
            file_list = zip_ref.namelist()
            
            # 1. Читаем общие строки (sharedStrings.xml)
            shared_strings = {}
            if 'xl/sharedStrings.xml' in file_list:
                with zip_ref.open('xl/sharedStrings.xml') as ss_file:
                    ss_tree = ET.parse(ss_file)
                    ss_root = ss_tree.getroot()
                    ns = {'si': 'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
                    for idx, si in enumerate(ss_root.findall('si', ns)):
                        text_elem = si.find('.//t', ns)
                        if text_elem is not None and text_elem.text:
                            shared_strings[idx] = text_elem.text
                        else:
                            text_parts = []
                            for t in si.iterfind('.//t', ns):
                                if t.text:
                                    text_parts.append(t.text)
                            if text_parts:
                                shared_strings[idx] = ''.join(text_parts)
            
            # 2. Читаем листы
            sheets_data = {}
            sheet_files = [f for f in file_list if f.startswith('xl/worksheets/sheet') and f.endswith('.xml')]
            
            if not sheet_files:
                print("❌ Листы не найдены в архиве.")
                return None
            
            for sheet_file in sorted(sheet_files):
                sheet_name = os.path.basename(sheet_file).replace('.xml', '')
                print(f"📄 Обработка листа: {sheet_name}...")
                rows_dict = {}
                
                with zip_ref.open(sheet_file) as xml_file:
                    tree = ET.parse(xml_file)
                    root = tree.getroot()
                    ns = {'sheet': 'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
                    
                    for cell in root.findall('.//sheet:c', ns):
                        ref = cell.get('r')
                        if not ref:
                            continue
                        row_num = get_cell_coordinate(ref)
                        value = None
                        cell_type = cell.get('t')
                        value_elem = cell.find('sheet:v', ns)
                        
                        if value_elem is not None and value_elem.text:
                            raw_value = value_elem.text
                            if cell_type == 's':
                                try:
                                    value = shared_strings.get(int(raw_value), raw_value)
                                except ValueError:
                                    value = raw_value
                            elif cell_type == 'b':
                                value = 'TRUE' if raw_value == '1' else 'FALSE'
                            elif cell_type == 'e':
                                value = f'#ERROR#{raw_value}'
                            else:
                                value = raw_value
                        
                        if value is None:
                            inline_str = cell.find('sheet:is/sheet:t', ns)
                            if inline_str is not None and inline_str.text:
                                value = inline_str.text
                        
                        if value is not None:
                            if row_num not in rows_dict:
                                rows_dict[row_num] = {}
                            col_label = ''.join(c for c in ref if c.isalpha())
                            rows_dict[row_num][col_label] = value
                
                if rows_dict:
                    max_row = max(rows_dict.keys())
                    all_cols = set()
                    for r in rows_dict.values():
                        all_cols.update(r.keys())
                    sorted_cols = sorted(list(all_cols), key=lambda x: (len(x), x))
                    
                    data = []
                    for r in range(1, max_row + 1):
                        row_data = [rows_dict.get(r, {}).get(c, '') for c in sorted_cols]
                        data.append(row_data)
                    
                    df = pd.DataFrame(data, columns=sorted_cols)
                    sheets_data[sheet_name] = df
                else:
                    sheets_data[sheet_name] = pd.DataFrame()
            
            return sheets_data
    
    except zipfile.BadZipFile:
        print("❌ Файл не является корректным ZIP-архивом (поврежден заголовок).")
    except Exception as e:
        print(f"❌ Ошибка при разборе XML: {e}")
        import traceback
        traceback.print_exc()
    
    return None

class RadaxParser:
    def __init__(self, download_dir=None, progress_callback=None):
        self.download_dir = download_dir or r"\\f12\exchange\Production\BD METALL do not delete\excel_avanz"
        self.progress_callback = progress_callback
        self.driver = None
        self._existing_files = set()
    
    def clear_download_folder(self, keep_pattern=None):
        """Очищает папку загрузок перед новой загрузкой"""
        try:
            if not os.path.exists(self.download_dir):
                self._log(f"📁 Создание папки: {self.download_dir}")
                os.makedirs(self.download_dir, exist_ok=True)
            
            deleted_count = 0
            for f in os.listdir(self.download_dir):
                if keep_pattern and re.match(keep_pattern, f):
                    continue
                if f.endswith(('.xlsx', '.xls', '.crdownload', '.tmp')):
                    file_path = os.path.join(self.download_dir, f)
                    try:
                        os.remove(file_path)
                        deleted_count += 1
                        self._log(f"🗑️ Удалён старый файл: {f}")
                    except Exception as e:
                        self._log(f"⚠️ Ошибка удаления {f}: {e}")
            
            self._log(f"✅ Очистка завершена. Удалено файлов: {deleted_count}")
        except Exception as e:
            self._log(f"⚠️ Ошибка очистки папки: {e}")
            
    def _log(self, message):
        if self.progress_callback:
            self.progress_callback(message)
    
    def _get_all_files(self):
        try:
            return [f for f in os.listdir(self.download_dir) if os.path.isfile(os.path.join(self.download_dir, f))]
        except:
            return []
    
    def _get_new_files(self):
        current_files = set(self._get_all_files())
        return list(current_files - self._existing_files)
    
    def _is_file_stable(self, filepath, check_interval=1.0, check_count=3):
        sizes = []
        for i in range(check_count):
            try:
                size = os.path.getsize(filepath)
                sizes.append(size)
                if len(sizes) > 1 and sizes[-1] != sizes[-2]:
                    return False
                time.sleep(check_interval)
            except:
                return False
        return True
    
    def check_network_access(self, url="http://f12/radax/"):
        try:
            urllib.request.urlopen(url, timeout=5)
            return True
        except:
            try:
                socket.create_connection(("f12", 80), timeout=5)
                return True
            except:
                return False
    
    def _get_driver(self):
        browsers = [
            ("Chrome", ChromeOptions, ChromeDriverManager, ChromeService),
            ("Edge", EdgeOptions, EdgeChromiumDriverManager, EdgeService),
            ("Firefox", FirefoxOptions, GeckoDriverManager, FirefoxService)
        ]
        for name, Options, DriverManager, Service in browsers:
            try:
                self._log(f"🔍 Попытка запуска {name}...")
                options = Options()
                options.add_argument("--headless=new")
                options.add_argument("--disable-blink-features=AutomationControlled")
                options.add_argument("--disable-gpu")
                options.add_argument("--no-sandbox")
                options.add_argument("--window-size=1920,1080")
                prefs = {
                    "download.default_directory": self.download_dir,
                    "download.prompt_for_download": False,
                    "download.directory_upgrade": True,
                    "safebrowsing.enabled": False,
                    "browser.download.folderList": 2,
                    "browser.helperApps.neverAsk.saveToDisk": "application/vnd.ms-excel,application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
                }
                options.add_experimental_option("prefs", prefs)
                options.add_experimental_option("excludeSwitches", ["enable-automation"])
                driver = webdriver.Chrome(
                    service=Service(DriverManager().install()),
                    options=options
                )
                self._log(f"✅ {name} успешно запущен")
                return driver
            except Exception as e:
                self._log(f"⚠️ {name} не доступен: {str(e)[:50]}")
                continue
        return None
    
    def login(self, username="AGR", password="2023", login_url="http://f12/radax/user/login"):
        self._log(f"🔐 Вход в систему: {login_url}")
        self.driver.get(login_url)
        time.sleep(2)
        try:
            user_field = WebDriverWait(self.driver, 10).until(
                EC.presence_of_element_located((By.NAME, "userid"))
            )
            user_field.clear()
            user_field.send_keys(username)
            pass_field = self.driver.find_element(By.NAME, "password")
            pass_field.clear()
            pass_field.send_keys(password)
            pass_field.submit()
            time.sleep(3)
            self._log("✅ Авторизация выполнена")

            self._log(f"❌ Ошибка авторизации: {e}")

    def set_date_range(self, start_date, end_date):
        start_str = start_date.strftime("%d/%m/%Y")
        end_str = end_date.strftime("%d/%m/%Y")
        self._log(f"📅 Установка дат: {start_str} — {end_str}")
        for element_id, value in [("start", start_str), ("end", end_str)]:
            try:
                input_element = WebDriverWait(self.driver, 10).until(
                    EC.presence_of_element_located((By.ID, element_id))
                )
                self.driver.execute_script("""
                    var element = arguments[0];
                    var value = arguments[1];
                    element.value = value;
                    ['input', 'change', 'blur'].forEach(function(eventName) {
                        var event = document.createEvent('HTMLEvents');
                        event.initEvent(eventName, true, false);
                        element.dispatchEvent(event);
                    });
                """, input_element, value)
                self._log(f"   ✅ Поле #{element_id} = {value}")
            except Exception as e:
                self._log(f"   ⚠️ Ошибка поля #{element_id}: {e}")
            time.sleep(1)
    
    def select_select2_option(self, container_id, option_title):
        try:
            select2_container = WebDriverWait(self.driver, 10).until(
                EC.element_to_be_clickable((By.ID, container_id))
            )

            option = WebDriverWait(self.driver, 10).until(
                EC.element_to_be_clickable(
                    (By.XPATH, f"//li[contains(@title, '{option_title}') or text()='{option_title}']")
                )

            option.click()
            time.sleep(0.5)
        except Exception as e:
            self._log(f"⚠️ Не выбрано {container_id}: {e}")

    def export_to_excel(self):
        self._log("📊 Экспорт в Excel...")
        try:
            excel_btn = WebDriverWait(self.driver, 10).until(
                EC.element_to_be_clickable(
                    (By.XPATH, "//span[text()='Excel']/.. | //button[contains(., 'Excel')]")
                )

            return False
    
    def _find_latest_excel_file(self):
        try:
            files = [f for f in os.listdir(self.download_dir)
                    if f.endswith(('.xlsx', '.xls')) and not f.endswith('.crdownload')]
            if not files:
                return None
            full_paths = [os.path.join(self.download_dir, f) for f in files]
            latest = max(full_paths, key=os.path.getctime)
            return latest
        except Exception as e:
            self._log(f"⚠️ Ошибка при поиске последнего файла: {e}")
            return None
    
    def wait_for_download(self, timeout=30, poll_interval=0.5):
        self._log("⏳ Ожидание завершения загрузки файла...")
        start_time = time.time()
        downloaded_file = None
        last_crdownload = None
        while time.time() - start_time < timeout:
            all_files = self._get_all_files()
            crdownload_files = [f for f in all_files if f.endswith('.crdownload')]
            excel_files = [f for f in all_files
                          if f.endswith(('.xlsx', '.xls')) and not f.endswith('.crdownload')]
            if crdownload_files:
                last_crdownload = crdownload_files[-1]
                self._log(f"   ⏳ Загрузка... ({int(time.time()-start_time)}/{timeout} сек)")
                time.sleep(poll_interval)
                continue
            if excel_files:
                latest_excel = max([os.path.join(self.download_dir, f) for f in excel_files],
                                  key=os.path.getctime)
                if self._is_file_stable(latest_excel, check_interval=0.3, check_count=2):
                    downloaded_file = os.path.basename(latest_excel)
                    self._log(f"   ✅ Файл готов: {downloaded_file}")
                    break
            time.sleep(poll_interval)
        if not downloaded_file and last_crdownload:
            old_path = os.path.join(self.download_dir, last_crdownload)
            base_name = last_crdownload.replace('.crdownload', '')
            new_path = os.path.join(self.download_dir, base_name)
            try:
                if os.path.exists(old_path):
                    os.rename(old_path, new_path)
                    downloaded_file = base_name
                    self._log(f"   ✅ Файл переименован: {downloaded_file}")
            except Exception as e:
                self._log(f"   ⚠️ Ошибка переименования: {e}")
        if downloaded_file:
            return os.path.join(self.download_dir, downloaded_file)
        return None
    
    def run(self, start_date, end_date, target_url="http://f12/radax/produzione/avafasi"):
        self._log("🧹 Очистка папки загрузок...")
        self.clear_download_folder() 
        self._existing_files = set(self._get_all_files())
        downloaded_file = None
        try:
            self._log("🚀 Запуск парсера...")
            if not self.check_network_access():
                self._log("❌ Нет доступа к серверу f12")
                return None
            self.driver = self._get_driver()
            if not self.driver:
                self._log("❌ Не удалось запустить ни один браузер")
                return None
            if not self.login():
                return None
            self._log(f"🌐 Переход на страницу: {target_url}")
            self.driver.get(target_url)
            time.sleep(2)
            self.set_date_range(start_date, end_date)
            time.sleep(1)
            self.select_select2_option("select2-darep-container", "[OFFICINA M] OFFICINA MECCANICA")
            self.select_select2_option("select2-arep-container", "[OFFICINA M] OFFICINA MECCANICA")
            self.select_select2_option("select2-dalav-container", "[TGL] TAGLIO LASER")
            self.select_select2_option("select2-alav-container", "[TGL] TAGLIO LASER")
            time.sleep(2)
            self.apply_filter()
            time.sleep(2)
            self.export_to_excel()
            downloaded_file = self.wait_for_download(timeout=30)
        except Exception as e:
            self._log(f"❌ Критическая ошибка: {e}")
            import traceback
            traceback.print_exc()
        finally:
            if self.driver:
                self.driver.quit()
                self._log("🏁 Браузер закрыт")
        if downloaded_file:
            self._log(f"✅ Файл: {downloaded_file}")
            return os.path.join(self.download_dir, downloaded_file)
        return None

        """Асинхронная загрузка данных для избежания зависания"""
        def load_task():
            # НЕ используем БД в этом потоке! Только JSON файлы
            self.load_data()
            self.integrate_excel_data()
            self.migrate_old_scrap()
            self.transfer_previous_month_totals()
            self.calculate_totals()
            return True
        
        def on_loaded():
            self.update_display()
            self.status_var.set("Готов")
            # ← БД подключаем ТОЛЬКО в главном потоке
            if self.db_path:
                self.db = Database()
                if self.db.connect(self.db_path):
                    self.records = self.db.load_all_records()
                    self._ensure_timestamps()
                    self.start_auto_sync()
                else:
                    self.db = None
            # Запускаем мигание индикаторов
            self.root.after(1500, self.update_blinking)
        
        # Загружаем в фоне
        thread = threading.Thread(target=load_task, daemon=True)
        thread.start()
        
        # После загрузки обновляем UI в главном потоке
        self.root.after(1500, on_loaded)

    def load_f12_data_cache(self):
        """Загружает и кэширует данные из F12 для быстрой проверки наличия данных."""
        # ✅ ЧИТАЕМ ИЗ БД ВМЕСТО JSON
        if self.db and self.db.connection:
            try:
                f12_data = self.db.load_f12_records()
                f12_cache = {}
                
                # Получаем артикулы заготовок из БД
                parts_articles, _ = self.db.load_roll_articles()
                self.part_areas_cache = dict(zip(parts_articles, _)) if _ else {}
                
                for record in f12_data:
                    date = record['date']
                    part_article = record['part_article']
                    if part_article in parts_articles:
                        # Кэшируем по дате для быстрого доступа
                        if date not in f12_cache:
                            f12_cache[date] = []
                        f12_cache[date].append(record)
                
                self.f12_data_cache = f12_cache
            except Exception as e:
                print(f"Ошибка при загрузке кэша F12 из БД: {e}")
                self.f12_data_cache = {}
        else:
            self.f12_data_cache = {}
    
    def has_f12_data_for_cell(self, date_str, article):
        if self.f12_data_cache is None:
            self.load_f12_data_cache()
        key = (date_str, article)
        return key in self.f12_data_cache and len(self.f12_data_cache[key]) > 0
    
    def get_glasswool_cell_color(self, article, date_str):
        if self.f12_data_cache is None:
            self.load_f12_data_cache()
        if not self.f12_data_cache or date_str not in self.f12_data_cache:
            return None
        day_records = self.f12_data_cache[date_str]
        parts_area = 0.0
        has_relevant_parts = False
        if self.db and self.db.connection:
            if not self.part_areas_cache:
                arts, areas = self.db.load_roll_articles()
                self.part_areas_cache = dict(zip(arts, areas))
            for rec in day_records:
                part_art = rec['part_article']
                if part_art in self.part_areas_cache:
                    good_qty = rec.get('good_qty', 0)
                    area_per_unit = self.part_areas_cache.get(part_art, 0)
                    parts_area += good_qty * area_per_unit
                    has_relevant_parts = True
        if not has_relevant_parts:
            return None
        roll_area = 0.0
        article_data = self.records.get(article, {})
        area_per_roll = article_data.get('area', 0)
        if date_str in article_data.get('days', {}):
            day_ops = article_data['days'][date_str]
            total_rolls = sum(day_ops.values())
            roll_area = total_rolls * area_per_roll
        if roll_area == 0:
            return self.COLOR_GLASS_DEFICIT
        if roll_area < parts_area:
            return self.COLOR_GLASS_DEFICIT
        return self.COLOR_GLASS_SURPLUS

    def fmt_number(self, value):
        if isinstance(value, float) and value.is_integer():
            return str(int(value))
        elif isinstance(value, (int, float)):
            s = f"{value:.1f}"
            if s.endswith('.0'):
                return s[:-2]
            return s
        else:
            return str(value)
    
    def generate_columns(self):
        columns = ['Артикул', 'Начальный остаток']
        days_in_month = calendar.monthrange(self.current_date.year, self.current_date.month)[1]
        week_counter = 1
        for day in range(1, days_in_month + 1):
            columns.append(str(day))
            date = datetime(self.current_date.year, self.current_date.month, day)
            if date.weekday() == 6 and day < days_in_month:
                columns.append(f'Итог {week_counter}')
                columns.append(f'Пересчет {week_counter}')
                week_counter += 1
        columns.append('Итог месяца')
        columns.append('Пересчет')
        return tuple(columns)
    
    def transfer_previous_month_totals(self):
        if self.current_date.month == 1:
            prev_year = self.current_date.year - 1
            prev_month = 12
        else:
            prev_year = self.current_date.year
            prev_month = self.current_date.month - 1
        prev_month_key = f"{prev_year}-{prev_month:02d}"
        for article in self.articles + self.summary_rows + self.glass_articles:
            if article in self.records:
                changed = False
                # Сначала проверяем пересчёт месяца - если есть, используем его
                if 'monthly_recalc' in self.records[article] and prev_month_key in self.records[article]['monthly_recalc']:
                    recalc_value = self.records[article]['monthly_recalc'][prev_month_key]
                    if recalc_value is not None and recalc_value != "":
                        self.records[article]['initial_balance'] = recalc_value
                        changed = True
                # Если нет пересчёта, используем итог месяца
                elif 'monthly_totals' in self.records[article] and prev_month_key in self.records[article]['monthly_totals']:
                    self.records[article]['initial_balance'] = self.records[article]['monthly_totals'][prev_month_key]
                    changed = True
                if changed:
                    self._update_timestamp(article)
                    self._save_to_db(article)
                    self.save_data()
    
    def load_articles_from_excel(self):
        file_path = filedialog.askopenfilename(
            title="Выберите файл Excel с данными из 1С:ERP",
            filetypes=[("Excel files", "*.xlsx *.xls"), ("All files", "*.*")]
        )
        if not file_path:
            return
        
        # === ИСПОЛЬЗУЕМ XML-ПАРСЕР ВМЕСТО pandas.read_excel ===
        sheets_data = extract_excel_data_raw(file_path)
        
        if not sheets_data:
            messagebox.showerror("Ошибка", "Не удалось прочитать файл через XML-парсер")
            return
        
        # Берём первый лист
        df = list(sheets_data.values())[0]
        
        if df.empty or df.shape[1] < 10:
            messagebox.showerror("Ошибка", "Файл не содержит необходимых столбцов (ожидается минимум 10).")
            return
        
        # Определяем колонки
        f12_col = None
        for col in df.columns:
            if 'Номенклатура рабочего центра' in str(col) or 'Номенклатура RADAX' in str(col):
                f12_col = col
                break
        
        if not f12_col:
            messagebox.showerror("Ошибка", "В файле нет столбцов 'Номенклатура рабочего центра' или 'Номенклатура RADAX'.")
            return
        
        required_columns = ['Номенклатура', 'Артикул', 'Длина', 'Ширина', 'Толщина', 'Плотность', 'Вид отхода ссылка']
        missing = [col for col in required_columns if col not in df.columns]
        if missing:
            messagebox.showerror("Ошибка", f"В файле отсутствуют столбцы: {', '.join(missing)}")
            return
        
        added_count = 0
        updated_count = 0
        errors = []
        
        for idx, row in df.iterrows():
            try:
                f12_art = str(row[f12_col]).strip()
                if pd.isna(f12_art) or f12_art == '':
                    continue
                name_1c = str(row['Номенклатура']).strip() if pd.notna(row['Номенклатура']) else ''
                article_A = str(row['Артикул']).strip() if pd.notna(row['Артикул']) else ''
                
                def to_float(val):
                    if pd.isna(val):
                        return 0.0
                    s = str(val).strip().replace(',', '.')
                    try:
                        return float(s)
                    except ValueError:
                        return 0.0
                
                length = to_float(row['Длина'])
                width = to_float(row['Ширина'])
                thickness = to_float(row['Толщина'])
                density = to_float(row['Плотность'])
                weight = 0.0
                if 'Вес' in df.columns:
                    weight = to_float(row['Вес'])
                scrap_material = str(row['Вид отхода ссылка']).strip() if pd.notna(row['Вид отхода ссылка']) else ''
                
                if f12_art not in self.articles:
                    self.articles.append(f12_art)
                    self.records[f12_art] = {
                        'initial_balance': 0,
                        'days': {},
                        'week_summaries': {},
                        'totals': {},
                        'monthly_totals': {},
                        'monthly_recalc': {},
                        'monthly_week_summaries': {},
                        'running_balances': {},
                        'min_balance': 0,
                        'info': {},
                        '_timestamp': time.time()
                    }
                    added_count += 1
                else:
                    updated_count += 1
                
                info = self.records[f12_art].get('info', {})
                info.update({
                    'article_A': article_A,
                    'name_1c': name_1c,
                    'length': length,
                    'width': width,
                    'thickness': thickness,
                    'weight': weight,
                    'density': density,
                    'scrap_material': scrap_material,
                    'weight_manual': False
                })
                self.records[f12_art]['info'] = info
                self._update_timestamp(f12_art)
                self._save_to_db(f12_art)
            except Exception as e:
                errors.append(f"Строка {idx+2}: {e}")
        
        self.save_data()
        self.calculate_totals()
        self.update_display()
        
        for child in self.root.winfo_children():
            if isinstance(child, ctk.CTkToplevel) and child.title() == "Управление артикулами":
                for widget in child.winfo_children():
                    if isinstance(widget, ttk.Treeview):
                        self.refresh_article_tree_in_manager(widget)
                    elif isinstance(widget, ctk.CTkFrame):
                        for sub in widget.winfo_children():
                            if isinstance(sub, ttk.Treeview):
                                self.refresh_article_tree_in_manager(sub)
                                break
        
        msg = f"Загрузка завершена.\nДобавлено артикулов: {added_count}\nОбновлено: {updated_count}"
        if errors:
            msg += f"\nОшибки в строках:\n" + "\n".join(errors[:5])
            if len(errors) > 5:
                msg += f"\n... и ещё {len(errors)-5}"
        messagebox.showinfo("Результат", msg)
    
    def migrate_old_scrap(self):
        for article in self.summary_rows:
            if article in self.records and 'scrap' in self.records[article]:
                for date_str, scrap_data in self.records[article]['scrap'].items():
                    if 'лом' in scrap_data:
                        if 'days' not in self.records[article]:
                            self.records[article]['days'] = {}
                        if date_str not in self.records[article]['days']:
                            self.records[article]['days'][date_str] = {}
                        if 'лом' not in self.records[article]['days'][date_str]:
                            self.records[article]['days'][date_str]['лом'] = scrap_data['лом']
                del self.records[article]['scrap']
                self._update_timestamp(article)
                self._save_to_db(article)
                self.save_data()

    def save_current_month_data(self):
        month_key = self.current_date.strftime('%Y-%m')
        for article in self.articles + self.summary_rows + self.glass_articles:
            if article in self.records:
                if 'week_summaries' in self.records[article]:
                    if 'monthly_week_summaries' not in self.records[article]:
                        self.records[article]['monthly_week_summaries'] = {}
                    self.records[article]['monthly_week_summaries'][month_key] = self.records[article]['week_summaries'].copy()
                # Сохраняем пересчёт месяца только если он есть (не пустой)
                if 'month_recalc' in self.records[article]:
                    recalc_value = self.records[article]['month_recalc']
                    if recalc_value is not None and recalc_value != "" and recalc_value != 0:
                        if 'monthly_recalc' not in self.records[article]:
                            self.records[article]['monthly_recalc'] = {}
                        self.records[article]['monthly_recalc'][month_key] = recalc_value
        self.save_data()
    
    def load_month_data(self):
        month_key = self.current_date.strftime('%Y-%m')
        for article in self.articles + self.summary_rows + self.glass_articles:
            if article in self.records:
                if 'monthly_week_summaries' in self.records[article] and month_key in self.records[article]['monthly_week_summaries']:
                    self.records[article]['week_summaries'] = self.records[article]['monthly_week_summaries'][month_key].copy()
                else:
                    self.records[article]['week_summaries'] = {}
                # Загружаем пересчёт месяца только если он есть в сохранённых данных
                if 'monthly_recalc' in self.records[article] and month_key in self.records[article]['monthly_recalc']:
                    recalc_value = self.records[article]['monthly_recalc'][month_key]
                    if recalc_value is not None and recalc_value != "" and recalc_value != 0:
                        self.records[article]['month_recalc'] = recalc_value
                    else:
                        self.records[article].pop('month_recalc', None)
                else:
                    self.records[article].pop('month_recalc', None)
    
    def load_data(self):
        if os.path.exists(self.data_file):
            try:
                with open(self.data_file, 'r', encoding='utf-8') as f:

                self.records = {}
        else:
            self.records = {}
        self.glass_articles = []
        for article, data in self.records.items():
            if data.get('type') == 'glass':

                    'initial_balance': 0,
                    'days': {},
                    'week_summaries': {},
                    'totals': {},
                    'monthly_totals': {},
                    'monthly_recalc': {},
                    'monthly_week_summaries': {},
                    'running_balances': {}
                }
        self._ensure_timestamps()

    def add_record(self):
        article = self.article_var.get()
        if not article:
            messagebox.showerror("Ошибка", "Выберите артикул")
            return
        if article == self.scrap_row:
            messagebox.showwarning("Предупреждение", f"Строка '{self.scrap_row}' является разделителем и не может быть изменена")
            return
        try:
            qty = int(float(self.qty_var.get())) if self.qty_var.get() else 0
            record_type = self.type_var.get()
            date = self.date_var.get()
            if article not in self.records:
                self.records[article] = {
                    'initial_balance': 0,
                    'days': {},
                    'week_summaries': {},
                    'totals': {},
                    'monthly_totals': {},
                    'monthly_recalc': {},
                    'monthly_week_summaries': {},
                    'running_balances': {}
                }
            if record_type == "начальный остаток":
                self.records[article]['initial_balance'] = qty
            else:
                if date not in self.records[article]['days']:
                    self.records[article]['days'][date] = {}
                self.records[article]['days'][date][record_type] = qty
            self._update_timestamp(article)
            self._save_to_db(article)
            self.save_data()
            self.calculate_totals()
            self.update_display()
            self.qty_var.set("")
            self.status_var.set(f"Запись добавлена: {article}")
        except ValueError:
            messagebox.showerror("Ошибка", "Введите целое число")
    
    def calculate_totals(self):
        month_key = self.current_date.strftime('%Y-%m')
        year = self.current_date.year
        month = self.current_date.month
        days_in_month = calendar.monthrange(year, month)[1]
        for article in self.articles + self.glass_articles + self.summary_rows:
            if article in self.records:
                running_balance = self.records[article].get('initial_balance', 0)
                if 'totals' not in self.records[article]:
                    self.records[article]['totals'] = {}
                if 'running_balances' not in self.records[article]:
                    self.records[article]['running_balances'] = {}
                week_totals = {}
                week_recalc_values = self.records[article].get('week_summaries', {})
                current_week = 1
                for day in range(1, days_in_month + 1):
                    date_str = f"{year}-{month:02d}-{day:02d}"
                    current_date = datetime(year, month, day)
                    if day > 1 and current_date.weekday() == 0:
                        prev_week_key = f"week{current_week-1}"
                        if prev_week_key in week_recalc_values:
                            running_balance = week_recalc_values[prev_week_key]
                    if date_str in self.records[article].get('days', {}):
                        day_data = self.records[article]['days'][date_str]
                        for op_type, qty in day_data.items():
                            if op_type in ('расход', 'лом'):
                                running_balance -= qty
                            elif op_type == 'приход':
                                running_balance += qty
                            elif op_type == 'пересчет':
                                running_balance = qty
                    self.records[article]['running_balances'][date_str] = running_balance
                    if current_date.weekday() == 6:
                        week_key = f"week{current_week}"
                        week_totals[week_key] = running_balance
                        if day < days_in_month:
                            current_week += 1
                last_day_date = datetime(year, month, days_in_month)
                if last_day_date.weekday() != 6:
                    week_key = f"week{current_week}"
                    week_totals[week_key] = running_balance
                self.records[article]['totals'] = week_totals
                self.records[article]['totals']['month'] = running_balance
                if 'monthly_totals' not in self.records[article]:
                    self.records[article]['monthly_totals'] = {}
                self.records[article]['monthly_totals'][month_key] = running_balance
        self.save_data()
        self.status_var.set("Итоги рассчитаны")
    
    def toggle_row_highlight(self, article):
        if article == self.scrap_row:
            return
        if article in self.highlighted_rows and self.highlighted_rows[article]:
            self.highlighted_rows[article] = False
        else:
            self.highlighted_rows[article] = True
        self.update_display()
    
    def is_row_highlighted(self, article):
        return self.highlighted_rows.get(article, False)
    
    def get_highlighted_bg_color(self, base_color):
        if base_color.startswith('#'):
            r = int(base_color[1:3], 16)
            g = int(base_color[3:5], 16)
            b = int(base_color[5:7], 16)
            r = min(255, int(r * 1.5))
            g = min(255, int(g * 1.5))
            b = min(255, int(b * 1.5))
            return f'#{r:02x}{g:02x}{b:02x}'
        return base_color

    def create_scrollable_scrap_row(self, row_idx, year, month, days_in_month):
        col_idx = 0
        row_height = 1
        label_empty = tk.Label(self.table_inner, text="", borderwidth=1, relief="solid",
                              width=20, height=row_height, bg='#5a3a3a')
        label_empty.grid(row=row_idx, column=col_idx, sticky="nsew", padx=0, pady=0)
        col_idx += 1
        week_total = 0
        current_week = 1
        for day in range(1, days_in_month + 1):
            date_str = f"{year}-{month:02d}-{day:02d}"
            current_day_date = datetime(year, month, day)
            day_scrap = 0
            for summary in self.summary_rows:
                if summary in self.records and date_str in self.records[summary].get('days', {}):
                    if 'лом' in self.records[summary]['days'][date_str]:
                        day_scrap += self.records[summary]['days'][date_str]['лом']
            week_total += day_scrap
            bg_color = '#5a3a3a'
            if current_day_date.date() == self.today and year == self.today.year and month == self.today.month:
                bg_color = '#704a4a'
            label = tk.Label(self.table_inner, text=str(day_scrap) if day_scrap > 0 else "",
                            borderwidth=1, relief="solid", width=8, height=row_height, bg=bg_color,

            label.grid(row=row_idx, column=col_idx, sticky="nsew", padx=0, pady=0)
            ToolTip(label, f"Лом за {day:02d}.{month:02d}: {day_scrap}")
            col_idx += 1
            if current_day_date.weekday() == 6 and day < days_in_month:
                label_total = tk.Label(self.table_inner, text=str(week_total) if week_total > 0 else "",

        label_month_recalc.grid(row=row_idx, column=col_idx, sticky="nsew", padx=0, pady=0)
    
    def get_month_scrap_total(self, year, month):
        totals = []
        days_in_month = calendar.monthrange(year, month)[1]
        for day in range(1, days_in_month + 1):
            date_str = f"{year}-{month:02d}-{day:02d}"
            day_total = 0
            for summary in self.summary_rows:
                if summary in self.records and date_str in self.records[summary].get('days', {}):
                    if 'лом' in self.records[summary]['days'][date_str]:
                        day_total += self.records[summary]['days'][date_str]['лом']
            totals.append(day_total)
        return totals

    def open_day_details(self, event, article, date_str):
        widget = event.widget
        cell_text = widget.cget("text")
        match = re.search(r'\(([\d.]+)\)', cell_text)
        if not match:
            messagebox.showerror("Ошибка", "Не удалось определить площадь рулонов в ячейке.")
            return
        roll_area = float(match.group(1))
        f12_data = []
        if self.db and self.db.connection:
            f12_data = self.db.load_f12_records(date_from=date_str, date_to=date_str)
        else:
            messagebox.showerror("Ошибка", "База данных не подключена")
            return

        day_records = f12_data  # Данные уже отфильтрованы по дате в методе БД

        if not day_records:
            messagebox.showinfo("Информация", f"Нет данных за {date_str}.")
            return
        day_records = [rec for rec in f12_data if rec['date'] == date_str]
        if not day_records:
            messagebox.showinfo("Информация", f"Нет данных за {date_str}.")
            return
        parts_articles, parts_areas = self.db.load_roll_articles()
        part_to_area = dict(zip(parts_articles, parts_areas))
        dialog = ctk.CTkToplevel(self.root)
        dialog.title(f"Детали за {date_str} – {article}")
        dialog.geometry("1300x550")
        dialog.transient(self.root)
        dialog.grab_set()
        columns = ('part_article', 'good_qty', 'defect_qty', 'part_name', 'order_num', 'area')
        tree = ttk.Treeview(dialog, columns=columns, show='headings', height=12)
        tree.heading('part_article', text='Артикул детали')
        tree.heading('good_qty', text='Кол-во годных')
        tree.heading('defect_qty', text='Брак')
        tree.heading('part_name', text='Наименование')
        tree.heading('order_num', text='Номер заказа')
        tree.heading('area', text='Площадь, м²')
        tree.column('part_article', width=150)
        tree.column('good_qty', width=80, anchor='center')
        tree.column('defect_qty', width=80, anchor='center')
        tree.column('part_name', width=250)
        tree.column('order_num', width=120)
        tree.column('area', width=100, anchor='center')
        scrollbar = ttk.Scrollbar(dialog, orient='vertical', command=tree.yview)
        tree.configure(yscrollcommand=scrollbar.set)
        tree.pack(side='left', fill='both', expand=True, padx=10, pady=10)
        scrollbar.pack(side='right', fill='y')
        def copy_selection(event):
            selected = tree.selection()
            if selected:
                values = tree.item(selected[0])['values']
                text = '\t'.join(str(v) for v in values)
                dialog.clipboard_clear()
                dialog.clipboard_append(text)
        tree.bind('<Control-c>', copy_selection)
        total_parts_area = 0.0
        for rec in day_records:
            part_art = rec['part_article']
            area_per_unit = part_to_area.get(part_art, 0)
            good = rec['good_qty']
            defect = rec['defect_qty']
            area_total = good * area_per_unit
            total_parts_area += area_total
            area_with_defect = (good + defect) * area_per_unit
            tree.insert('', tk.END, values=(
                part_art,
                good,
                defect,
                rec['part_name'],
                rec['order_num'],
                f"{area_total:.3f}"
            ))
        tree.insert('', tk.END, values=("", "", "", "", "ИТОГО:", f"{total_parts_area:.3f}"), tags=('bold',))
        tree.tag_configure('bold', font=('Segoe UI', 9, 'bold'))
        remainder = roll_area - total_parts_area
        bottom_frame = ctk.CTkFrame(dialog)
        bottom_frame.pack(fill='x', padx=10, pady=10)
        left_frame = ctk.CTkFrame(bottom_frame, fg_color="transparent")
        left_frame.pack(side='left', padx=10)
        if remainder >= 0:
            remainder_text = f"✓ Остаток: {remainder:.3f} м²"
            remainder_color = "green"
        else:
            remainder_text = f"✗ Не хватает: {abs(remainder):.3f} м²"
            remainder_color = "red"
        lbl_remainder = ctk.CTkLabel(left_frame, text=remainder_text, font=("Segoe UI", 12, "bold"),
                                    text_color=remainder_color)
        lbl_remainder.pack(side='top', padx=10, pady=5)
        right_frame = ctk.CTkFrame(bottom_frame, fg_color="transparent")
        right_frame.pack(side='right', padx=10)
        def save_report():
            default_name = f"Стекловата_{datetime.now().strftime('%d.%m.%Y')}.txt"

    def sync_with_db(self):
        if not self.db or not self.db.connection:
            messagebox.showerror("Ошибка", "База данных не подключена.\nВыберите файл БД в настройках.")
            return
        db_records = self.db.load_all_records()
        if db_records is None:
            messagebox.showerror("Ошибка", "Не удалось загрузить данные из БД.")
            return
        for article, data in db_records.items():
            if '_timestamp' not in data:
                data['_timestamp'] = 0
        all_articles = set(self.records.keys()) | set(db_records.keys())
        for article in all_articles:
            local = self.records.get(article)
            remote = db_records.get(article)
            if local is None:
                self.records[article] = remote.copy()
                self.records[article]['_timestamp'] = time.time()
                self.unsaved_changes = True
            elif remote is None:
                self.db.save_article(article, local)
            else:
                local_ts = local.get('_timestamp', 0)
                remote_ts = remote.get('_timestamp', 0)
                if local_ts > remote_ts:
                    self.db.save_article(article, local)
                elif remote_ts > local_ts:
                    self.records[article] = remote.copy()
                    self.records[article]['_timestamp'] = time.time()
                    self.unsaved_changes = True
        self.glass_articles = []
        for article, data in self.records.items():
            if data.get('type') == 'glass':
                self.glass_articles.append(article)
        self.glass_articles.sort()
        self.calculate_totals()
        self.update_display()
        self.unsaved_changes = False
        self.status_var.set("Синхронизация с БД завершена")
    
    def start_auto_sync(self):

    def auto_sync(self):
        if not self.syncing and self.db and self.db.connection:
            self.syncing = True
            self.sync_with_db()
            self.syncing = False
        self.root.after(30000, self.auto_sync)

                self.db.save_all_records(self.records)
                self.save_data()
        self.backup_database()
        self.root.destroy()

    def rename_selected_article(self):
        selected = self.tree.selection()

            return

        rename_dialog = ctk.CTkToplevel(self.root)
        rename_dialog.title("Переименовать артикул")

        rename_dialog.transient(self.root)
        rename_dialog.grab_set()
        ctk.CTkLabel(rename_dialog, text=f"Старое название: {old_name}\nВведите новое название:").pack(pady=10)

        entry.pack(pady=5)
        entry.focus()
        def save_rename():
            new_name = new_var.get().strip()
            if not new_name:
                messagebox.showerror("Ошибка", "Название не может быть пустым")
                return
            if new_name == old_name:
                rename_dialog.destroy()

    def _perform_clear_month_data(self):
        month_key = self.current_date.strftime('%Y-%m')
        for article in self.articles + self.summary_rows + self.glass_articles:
            if article in self.records:
                self.records[article]['days'] = {}
                self.records[article]['week_summaries'] = {}
                self.records[article]['totals'] = {}
                self.records[article]['running_balances'] = {}
                if 'month_recalc' in self.records[article]:
                    del self.records[article]['month_recalc']
                if 'monthly_totals' in self.records[article] and month_key in self.records[article]['monthly_totals']:
                    del self.records[article]['monthly_totals'][month_key]
                if 'monthly_recalc' in self.records[article] and month_key in self.records[article]['monthly_recalc']:
                    del self.records[article]['monthly_recalc'][month_key]
                if 'monthly_week_summaries' in self.records[article] and month_key in self.records[article]['monthly_week_summaries']:
                    del self.records[article]['monthly_week_summaries'][month_key]
                self._update_timestamp(article)
                self._save_to_db(article)
                self.save_data()
                self.calculate_totals()
                self.update_display()
                self.status_var.set(f"Данные за {month_key} очищены")
        messagebox.showinfo("Успех", f"Данные за {month_key} успешно очищены")

    def get_final_month_value(self, article):
        data = self.records.get(article, {})
        if 'month_recalc' in data:
            return data['month_recalc']
        return data.get('totals', {}).get('month', 0)

    def get_indicator_state(self, article):
        min_bal = self.records.get(article, {}).get('min_balance', 0)
        if min_bal == 0:
            return 'grey'
        final = self.get_final_month_value(article)
        if final < min_bal:
            return 'red'
        elif final <= min_bal * 1.3:
            return 'yellow'
        else:
            return 'grey'

    def update_blinking(self):
        if not hasattr(self, 'blink_state'):
            self.blink_state = False
        self.blink_state = not self.blink_state
        for article, indicator in self.indicator_labels.items():
            if article not in self.records:
                continue
            state = self.get_indicator_state(article)
            if state == 'grey':
                indicator.config(fg='gray')
            else:
                if self.blink_state:
                    indicator.config(fg=state)
                else:
                    indicator.config(fg='gray')
        self.root.after(1500, self.update_blinking)

            month_scrap = sum(self.get_month_scrap_total(year, month))

            for summary in self.summary_rows:

    def open_min_balance_editor(self):
        dialog = ctk.CTkToplevel(self.root)
        dialog.title("Минимальные остатки")
        dialog.geometry("500x800")
        dialog.transient(self.root)
        dialog.grab_set()
        header_frame = ctk.CTkFrame(dialog)
        header_frame.pack(fill="x", padx=10, pady=5)
        ctk.CTkLabel(header_frame, text="Артикул", font=("Segoe UI", 12, "bold"), width=300).pack(side="left", padx=5)
        ctk.CTkLabel(header_frame, text="Мин. остаток", font=("Segoe UI", 12, "bold"), width=100).pack(side="left", padx=5)
        scroll_frame = ctk.CTkScrollableFrame(dialog)
        scroll_frame.pack(fill="both", expand=True, padx=10, pady=5)
        entries = {}
        all_arts = self.articles + self.glass_articles
        for i, article in enumerate(all_arts):
            row_frame = ctk.CTkFrame(scroll_frame)
            row_frame.pack(fill="x", pady=1)
            lbl = ctk.CTkLabel(row_frame, text=article, anchor="w", width=300)
            lbl.pack(side="left", padx=5)
            current_val = self.records.get(article, {}).get('min_balance', 0)
            var = tk.StringVar(value=str(current_val))
            entry = ctk.CTkEntry(row_frame, textvariable=var, width=100)
            entry.pack(side="left", padx=5)
            entries[article] = var
        btn_frame = ctk.CTkFrame(dialog)
        btn_frame.pack(fill="x", padx=10, pady=10)
        def save():
            for article, var in entries.items():
                try:
                    val = int(var.get())
                except ValueError:
                    val = 0
                if article in self.records:
                    self.records[article]['min_balance'] = val
                    self._update_timestamp(article)
                    self._save_to_db(article)
            self.save_data()
            self.calculate_totals()
            self.update_display()
            dialog.destroy()
        def cancel():
            dialog.destroy()
        ctk.CTkButton(btn_frame, text="Сохранить", command=save, width=100).pack(side="left", padx=5)
        ctk.CTkButton(btn_frame, text="Отмена", command=cancel, width=100).pack(side="left", padx=5)
    
    def get_final_month_value(self, article):
        data = self.records.get(article, {})
        if 'month_recalc' in data:
            return data['month_recalc']
        return data.get('totals', {}).get('month', 0)
    
    def get_indicator_state(self, article):
        min_bal = self.records.get(article, {}).get('min_balance', 0)
        if min_bal == 0:
            return 'grey'
        final = self.get_final_month_value(article)
        if final < min_bal:
            return 'red'
        elif final <= min_bal * 1.3:
            return 'yellow'
        else:
            return 'grey'
    
    def update_blinking(self):
        if not hasattr(self, 'blink_state'):
            self.blink_state = False
        self.blink_state = not self.blink_state
        for article, indicator in self.indicator_labels.items():
            if article not in self.records:
                continue
            state = self.get_indicator_state(article)
            if state == 'grey':
                indicator.config(fg='gray')
            else:
                if self.blink_state:
                    indicator.config(fg=state)
                else:
                    indicator.config(fg='gray')
        self.root.after(1500, self.update_blinking)
    
    def open_roll_calculator(self):
        if not self.db or not self.db.connection:
            messagebox.showerror("Ошибка", "База данных не подключена.\nСначала выберите файл БД в настройках.")
            return
        RollCalculatorApp(parent=self.root, db=self.db, main_app=self, f12_file_path=self.f12_file)

class GlasswoolManager:
    def __init__(self, parent, app):

    def setup_ui(self):

        main_frame.pack(fill="both", expand=True, padx=10, pady=10)
        list_frame = ctk.CTkFrame(main_frame)
        list_frame.pack(fill="both", expand=True)
        scrollbar = ctk.CTkScrollbar(list_frame)
        scrollbar.pack(side="right", fill="y")
        self.listbox = tk.Listbox(list_frame, yscrollcommand=scrollbar.set, font=('Segoe UI', 12),
                                 bg='#2b2b2b', fg='white', selectbackground='#1f5f9f')

        btn_frame.pack(fill="x", padx=10, pady=10)

        ctk.CTkButton(btn_frame, text="Закрыть", command=self.window.destroy, width=100).pack(side="right", padx=5)
    
    def refresh_list(self):
        self.listbox.delete(0, tk.END)
        for article in self.app.glass_articles:
            data = self.app.records.get(article, {})
            if data.get('type') == 'glass':
                length = data.get('length', 0)
                width = data.get('width', 0)
                area = data.get('area', 0)
                display = f"{article}  (длина {self.app.fmt_number(length)} м, ширина {self.app.fmt_number(width)} м, площадь {self.app.fmt_number(area)} м²)"
                self.listbox.insert(tk.END, display)
    
    def add_article(self):
        dialog = GlasswoolDialog(self.window, "Добавление стекловаты", app=self.app)
        if dialog.result:
            length, width = dialog.result
            area = length * width
            name = f"Стекловата {self.app.fmt_number(length)}x{self.app.fmt_number(width)} ({self.app.fmt_number(area)} м²)"
            if name in self.app.records:
                messagebox.showerror("Ошибка", "Артикул с такими размерами уже существует!")
                return
            self.app.records[name] = {
                'type': 'glass',
                'length': length,
                'width': width,
                'area': area,
                'initial_balance': 0,
                'days': {},
                'week_summaries': {},
                'totals': {},
                'monthly_totals': {},
                'monthly_recalc': {},
                'monthly_week_summaries': {},
                'running_balances': {},
                'min_balance': 0,
                '_timestamp': time.time()
            }
            self.app.glass_articles.append(name)
            self.app.glass_articles.sort()
            self.app._save_to_db(name)
            self.app.save_data()
            self.app.update_display()
            self.refresh_list()
            self.app.refresh_data_display()
    
    def edit_article(self):
        selection = self.listbox.curselection()
        if not selection:
            messagebox.showinfo("Информация", "Выберите артикул для редактирования")
            return
        idx = selection[0]
        old_name = self.app.glass_articles[idx]
        data = self.app.records[old_name]
        old_length = data.get('length', 0)
        old_width = data.get('width', 0)
        dialog = GlasswoolDialog(self.window, "Редактирование стекловаты", length=old_length, width=old_width, app=self.app)
        if dialog.result:
            new_length, new_width = dialog.result
            new_area = new_length * new_width
            new_name = f"Стекловата {self.app.fmt_number(new_length)}x{self.app.fmt_number(new_width)} ({self.app.fmt_number(new_area)} м²)"
            if new_name != old_name and new_name in self.app.records:
                messagebox.showerror("Ошибка", "Артикул с такими размерами уже существует!")
                return
            old_data = self.app.records.pop(old_name)
            old_data['length'] = new_length
            old_data['width'] = new_width
            old_data['area'] = new_area
            old_data['_timestamp'] = time.time()
            self.app.records[new_name] = old_data
            self.app.glass_articles[idx] = new_name
            self.app.glass_articles.sort()
            if self.app.db and self.app.db.connection:
                self.app.db.delete_article(old_name)
                self.app.db.save_article(new_name, old_data)
            self.app.save_data()
            self.app.update_display()
            self.refresh_list()
            self.app.refresh_data_display()
    
    def delete_article(self):
        selection = self.listbox.curselection()
        if not selection:
            messagebox.showinfo("Информация", "Выберите артикул для удаления")
            return
        if not messagebox.askyesno("Подтверждение", "Удалить выбранный артикул стекловаты? Это удалит все данные по нему."):
            return
        idx = selection[0]
        article = self.app.glass_articles[idx]
        if self.app.db and self.app.db.connection:
            self.app.db.delete_article(article)
        del self.app.records[article]
        del self.app.glass_articles[idx]
        self.app.save_data()
        self.app.update_display()
        self.refresh_list()
        self.app.refresh_data_display()

        for row in self.tree.get_children():
            self.tree.delete(row)
        for art, area in zip(self.app.article_list, self.app.area_list):
            self.tree.insert('', 'end', values=(art, f"{area:.3f}"))

        dialog = ArticleDialog(self.window, "Добавление артикула")
        if dialog.result:
            name, area = dialog.result

                return

            self.refresh_list()

            messagebox.showinfo("Информация", "Выберите артикул для редактирования")
            return

        values = item['values']
        old_name = values[0]
        old_area = float(values[1])
        idx = self.app.article_list.index(old_name)

        art_name = item['values'][0]

        del self.app.article_list[idx]
        del self.app.area_list[idx]
        self.app.save_articles_to_db()
        self.app.update_display()
        self.refresh_list()

    def _run_parser_with_progress(self, date_from, date_to):
        progress_window = ProgressDialog(self.window, "Автоматическая загрузка из F12")
        download_dir = r"\\f12\exchange\Production\BD METALL do not delete\excel_avanz"
        parser = RadaxParser(download_dir=download_dir, progress_callback=progress_window.log)
        def parser_thread():
            file_path = parser.run(date_from, date_to)
            self.window.after(0, lambda: self._on_parser_complete(file_path, progress_window))
        thread = threading.Thread(target=parser_thread, daemon=True)
        thread.start()
    
    def _on_parser_complete(self, file_path, progress_window):
        progress_window.log("✅ Загрузка завершена")
        if file_path:
            file_path = self._wait_for_file_ready(file_path, progress_window)
        if not file_path or not os.path.exists(file_path):
            progress_window.log("🔍 Поиск последнего Excel-файла в папке...")
            file_path = self._find_latest_excel_in_folder(progress_window)
        if not file_path or not os.path.exists(file_path):
            progress_window.finish()
            self.window.after(2000, progress_window.destroy)
            messagebox.showerror("Ошибка", "Не удалось найти загруженный файл Excel.")
            return
        progress_window.log(f"✅ Файл найден: {os.path.basename(file_path)}")
        progress_window.log("⏳ Финализация файла...")
        time.sleep(2)
        progress_window.finish()
        self.window.after(1000, progress_window.destroy)
        self.window.after(1500, lambda: self._process_f12_file(file_path))
    
    def _wait_for_file_ready(self, file_path, progress_window, max_wait=20):
        import time
        if file_path.endswith('.crdownload'):
            progress_window.log("⏳ Ожидание завершения загрузки...")
            base_path = file_path.replace('.crdownload', '')
            last_size = -1
            for i in range(max_wait):
                time.sleep(1)
                if os.path.exists(base_path):
                    try:
                        current_size = os.path.getsize(base_path)
                        if current_size == last_size and current_size > 0:
                            progress_window.log(f"   ✅ Файл готов и стабилен: {os.path.basename(base_path)}")
                            return base_path
                        last_size = current_size
                        progress_window.log(f"   ⏳ Запись... {current_size} байт")
                    except:
                        pass
                elif os.path.exists(file_path):
                    try:
                        size = os.path.getsize(file_path)
                        progress_window.log(f"   ⏳ Загрузка... {size} байт ({i+1}/{max_wait} сек)")
                    except:
                        pass
                try:
                    if os.path.exists(file_path) and not os.path.exists(base_path):
                        os.rename(file_path, base_path)
                        progress_window.log("   ✅ Файл переименован принудительно")
                        time.sleep(1)
                        return base_path
                except Exception as e:
                    progress_window.log(f"   ⚠️ Ошибка переименования: {e}")
                if os.path.exists(base_path):
                    return base_path
        else:
            for i in range(5):
                time.sleep(0.5)
                if not os.path.exists(file_path):
                    break
            return file_path
        return None
    
    def _find_latest_excel_in_folder(self, progress_window):
        download_dir = r"\\f12\exchange\Production\BD METALL do not delete\excel_avanz"
        try:
            if not os.path.exists(download_dir):
                progress_window.log(f"❌ Папка не доступна: {download_dir}")
                return None
            excel_files = []
            for f in os.listdir(download_dir):
                if f.endswith(('.xlsx', '.xls')) and not f.endswith('.crdownload'):
                    full_path = os.path.join(download_dir, f)
                    excel_files.append((full_path, os.path.getmtime(full_path)))
            if not excel_files:
                progress_window.log("❌ Excel файлы не найдены")
                return None
            excel_files.sort(key=lambda x: x[1], reverse=True)
            latest_path = excel_files[0][0]
            progress_window.log(f"✅ Найден файл: {os.path.basename(latest_path)}")
            return latest_path
        except Exception as e:
            progress_window.log(f"❌ Ошибка поиска: {e}")
            return None
    
    def _process_f12_file(self, file_path):
        """ОБНОВЛЁННЫЙ МЕТОД: Использует XML-парсер вместо pandas.read_excel"""
        self.load_articles_from_db()
        
        # === ИСПОЛЬЗУЕМ XML-ПАРСЕР ДЛЯ ЧТЕНИЯ ===
        sheets_data = extract_excel_data_raw(file_path)
        
        if not sheets_data:
            messagebox.showerror("Ошибка", "Не удалось прочитать файл через XML-парсер")
            return
        
        # Берём первый лист
        df = list(sheets_data.values())[0]
        
        if df is None or df.empty or df.shape[1] < 10:
            messagebox.showerror("Ошибка", "Файл не содержит необходимых столбцов (ожидается минимум 10).")
            return
        
        json_file = self.f12_file_path
        if os.path.exists(json_file):
            try:
                with open(json_file, 'r', encoding='utf-8') as f:
                    existing_data = json.load(f)
            except:
                existing_data = []
        else:
            existing_data = []
        
        data_dict = {}
        for record in existing_data:
            key = (record['date'], record['part_article'])
            data_dict[key] = record
        
        processed = 0
        skipped = 0
        
        for idx, row in df.iterrows():
            try:
                date_str_raw = str(row.iloc[0]).strip()
                if pd.isna(date_str_raw) or not date_str_raw:
                    continue
                date_obj = None
                for fmt in ("%d.%m.%Y", "%d/%m/%Y", "%d.%m.%y", "%d/%m/%y"):
                    try:
                        date_obj = datetime.strptime(date_str_raw, fmt)
                        break
                    except ValueError:
                        continue
                if date_obj is None:
                    skipped += 1
                    continue
                date = date_obj.strftime("%Y-%m-%d")
                order_num = str(row.iloc[4]).strip() if not pd.isna(row.iloc[4]) else ""
                part_article_raw = str(row.iloc[5]).strip() if not pd.isna(row.iloc[5]) else ""
                if not part_article_raw:
                    continue
                part_article = re.sub(r'\s+', ' ', part_article_raw).strip().upper()
                part_name = str(row.iloc[6]).strip() if not pd.isna(row.iloc[6]) else ""
                good_str = str(row.iloc[8]).strip() if not pd.isna(row.iloc[8]) else "0"
                try:
                    good_qty = int(float(good_str))
                except:
                    good_qty = 0
                defect_str = str(row.iloc[9]).strip() if not pd.isna(row.iloc[9]) else "0"
                try:
                    defect_qty = int(float(defect_str))
                except:
                    defect_qty = 0
                if part_article not in self.article_list:
                    skipped += 1
                    continue
                key = (date, part_article)
                if key in data_dict:
                    old = data_dict[key]
                    old['good_qty'] = old.get('good_qty', 0) + good_qty
                    old['defect_qty'] = old.get('defect_qty', 0) + defect_qty
                    old['part_name'] = part_name
                    old['order_num'] = order_num
                else:
                    new_record = {
                        'date': date,
                        'part_article': part_article,
                        'part_name': part_name,
                        'good_qty': good_qty,
                        'defect_qty': defect_qty,
                        'order_num': order_num
                    }
                    data_dict[key] = new_record
                processed += 1
            except Exception as e:
                skipped += 1
                continue
        
        updated_data = list(data_dict.values())
        saved_count = 0
        for key, record in data_dict.items():
            # Сохраняем каждую запись в БД вместо JSON
            if self.db and self.db.connection:
                self.db.save_f12_record(
                    record['date'],
                    record['part_article'],
                    record['part_name'],
                    record['order_num'],
                    record['good_qty'],
                    record['defect_qty']
                )
                saved_count += 1

        # Кэш главного приложения нужно сбросить, чтобы он увидел новые данные
        if self.main_app:
            self.main_app.f12_data_cache = None

        messagebox.showinfo(
            "Загрузка завершена",
            f"Обработано записей: {processed}\n"
            f"Пропущено (не найдены в списке деталей): {skipped}\n"
            f"Сохранено в БД: {saved_count}"
        )
        
        messagebox.showinfo(
            "Загрузка завершена",
            f"Обработано записей: {processed}\n"
            f"Пропущено (не найдены в списке деталей): {skipped}"
        )
        self.main_app.f12_data_cache = None

        self.entry_quantity.delete(0, tk.END)
        self.entry_quantity.insert(0, "1")
        self.update_display()
    
    def remove_selected(self):
        if not self.listbox_selected.curselection():
            return
        idx = self.listbox_selected.curselection()[0]
        del self.selected_articles[idx]
        del self.selected_qtys[idx]
        self.update_display()
    
    def calculate(self):
        if not self.selected_articles:
            return
        used_area = 0.0
        for i, art in enumerate(self.selected_articles):
            try:
                art_idx = self.article_list.index(art)
                used_area += self.area_list[art_idx] * self.selected_qtys[i]
            except ValueError:
                continue
        result_area = self.total_area - used_area
        for row in self.tree.get_children():
            self.tree.delete(row)
        for i, art in enumerate(self.selected_articles):
            art_idx = self.article_list.index(art)
            item_area = self.area_list[art_idx] * self.selected_qtys[i]
            self.tree.insert('', tk.END, values=(art, self.selected_qtys[i], f"{item_area:.3f}"))
        self.tree.insert('', tk.END, values=("", "ИТОГО:", f"{used_area:.3f}"), tags=('bold',))
        self.tree.tag_configure('bold', font=('Segoe UI', 9, 'bold'))
        if result_area >= 0:
            self.label_total_area.configure(
                text=f"✓ Остаток: {result_area:.3f} м²",
                text_color="green"
            )
        else:
            self.label_total_area.configure(
                text=f"✗ Не хватает: {abs(result_area):.3f} м²",
                text_color="red"
            )

    def _process_export(self, article_display, date_from, date_to):
        candidate = article_display.split('(')[0].strip()
        glass_article = None
        for art, data in self.main_app.records.items():
            if data.get('type') == 'glass':
                if art == candidate or article_display.startswith(art):
                    glass_article = art
                    break
        if not glass_article:
            glass_list = [a for a, d in self.main_app.records.items() if d.get('type') == 'glass']
            messagebox.showerror(
                "Ошибка",
                f"Не найден артикул '{candidate}' в данных.\n"
                f"Доступные артикулы стекловаты:\n" + "\n".join(glass_list)
            )
            return
        roll_area = self.main_app.records[glass_article].get('area', 0)
        if roll_area <= 0:
            messagebox.showerror("Ошибка", f"Площадь рулона для '{glass_article}' не определена.")
            return
        f12_data = []
        if self.db and self.db.connection:
            f12_data = self.db.load_f12_records(date_from=date_from, date_to=date_to)
        else:
            messagebox.showerror("Ошибка", "База данных не подключена")
            return

        if not f12_data:
            messagebox.showinfo("Информация", "Нет данных F12 за выбранный период")
            return
        parts_articles, parts_areas = self.db.load_roll_articles()
        part_to_area = dict(zip(parts_articles, parts_areas))
        f12_by_date = {}
        for rec in f12_data:
            rec_date = rec.get('date', '')
            if date_from <= rec_date <= date_to:
                if rec_date not in f12_by_date:
                    f12_by_date[rec_date] = []
                f12_by_date[rec_date].append(rec)
        glass_record = self.main_app.records.get(glass_article, {})
        day_operations = glass_record.get('days', {})
        days_with_ops_no_f12 = []
        start = datetime.strptime(date_from, "%Y-%m-%d")
        end = datetime.strptime(date_to, "%Y-%m-%d")
        current = start
        while current <= end:
            date_str = current.strftime("%Y-%m-%d")
            if date_str in day_operations:
                if date_str not in f12_by_date:
                    days_with_ops_no_f12.append(date_str)
            current += timedelta(days=1)
        if days_with_ops_no_f12:
            msg = "Следующие дни имеют операции по стекловате, но нет данных F12:\n" + "\n".join(days_with_ops_no_f12)
            messagebox.showwarning("Внимание", msg)
        if not f12_by_date:
            messagebox.showinfo("Информация", "Нет данных F12 за выбранный период")
            return
        wb = Workbook()
        ws1 = wb.active
        ws1.title = "Детализация по дням"
        ws2 = wb.create_sheet("Итог")
        from openpyxl.styles import PatternFill, Font, Alignment, Fill
        header_font = Font(bold=True, size=11)
        header_fill = PatternFill(fill_type="solid", start_color="3D616B", end_color="3D616B")
        yellow_fill = PatternFill(fill_type="solid", start_color="FFFF00", end_color="FFFF00")
        no_fill = PatternFill(fill_type="none")
        headers1 = ['Дата', 'Артикул стекловаты', 'Суммарное кол-во рулонов за день (шт)',
                   'Суммарная площадь рулонов (м²)', 'Артикул детали', 'Кол-во годных', 'Брак',
                   'Площадь деталей (м²)', 'Остаток (м²)', 'Статус']
        for col, h in enumerate(headers1, 1):
            cell = ws1.cell(row=1, column=col, value=h)
            cell.font = header_font
            cell.fill = header_fill
            cell.alignment = Alignment(horizontal='center', vertical='center')
            ws1.column_dimensions[get_column_letter(col)].width = 18
        row_num = 2
        total_remainder = 0.0
        dates_sorted = sorted(f12_by_date.keys())
        for date_str in dates_sorted:
            day_f12 = f12_by_date[date_str]
            ops = day_operations.get(date_str, {})
            total_rolls = ops.get('приход', 0) + ops.get('расход', 0) + ops.get('пересчет', 0) + ops.get('лом', 0)
            area_rolls = total_rolls * roll_area
            day_parts_area = 0.0
            for rec in day_f12:
                part_art = rec['part_article']
                if part_art not in part_to_area:
                    continue
                good_qty = rec.get('good_qty', 0)
                part_area = part_to_area[part_art]
                day_parts_area += good_qty * part_area
            day_remainder = area_rolls - day_parts_area
            total_remainder += day_remainder
            status = "Остаток" if day_remainder >= 0 else "Не хватает"
            first_row = True
            for rec in day_f12:
                part_art = rec['part_article']
                good_qty = rec.get('good_qty', 0)
                defect_qty = rec.get('defect_qty', 0)
                part_area = part_to_area.get(part_art, 0)
                detail_area = good_qty * part_area
                if first_row:
                    ws1.cell(row=row_num, column=1, value=date_str)
                    ws1.cell(row=row_num, column=2, value=glass_article)
                    ws1.cell(row=row_num, column=3, value=total_rolls)
                    ws1.cell(row=row_num, column=4, value=round(area_rolls, 3))
                    ws1.cell(row=row_num, column=9, value=round(day_remainder, 3))
                    ws1.cell(row=row_num, column=10, value=status)
                    first_row = False
                ws1.cell(row=row_num, column=5, value=part_art)
                ws1.cell(row=row_num, column=6, value=good_qty)
                ws1.cell(row=row_num, column=7, value=defect_qty)
                ws1.cell(row=row_num, column=8, value=round(detail_area, 3))
                if status == "Не хватает":
                    for col in range(1, 11):
                        cell = ws1.cell(row=row_num, column=col)
                        cell.fill = PatternFill(fill_type="solid", start_color="FFFF00", end_color="FFFF00")
                row_num += 1
        ws1.cell(row=row_num, column=1, value="ИТОГО:")
        ws1.cell(row=row_num, column=9, value=round(total_remainder, 3))
        ws1.cell(row=row_num, column=9).font = Font(bold=True)
        ws2.cell(row=1, column=1, value="Период")
        ws2.cell(row=1, column=2, value="Артикул")
        ws2.cell(row=1, column=3, value="Общий остаток (м²)")
        ws2.cell(row=1, column=4, value="Статус")
        for col in range(1, 5):
            cell = ws2.cell(row=1, column=col)
            cell.font = header_font
            cell.fill = header_fill
            cell.alignment = Alignment(horizontal='center')
        final_status = "Остаток" if total_remainder >= 0 else "Не хватает"
        ws2.cell(row=2, column=1, value=f"{date_from} - {date_to}")
        ws2.cell(row=2, column=2, value=glass_article)
        ws2.cell(row=2, column=3, value=round(total_remainder, 3))
        ws2.cell(row=2, column=4, value=final_status)
        if final_status == "Не хватает":
            for col in range(1, 5):
                cell = ws2.cell(row=2, column=col)
                cell.fill = PatternFill(fill_type="solid", start_color="FFFF00", end_color="FFFF00")
        filename = f"Выгрузка стекловаты {date_from.replace('-', '_')}-{date_to.replace('-', '_')}.xlsx"
        file_path = filedialog.asksaveasfilename(
            defaultextension=".xlsx",
            filetypes=[("Excel files", "*.xlsx")],
            initialfile=filename,
            title="Сохранить выгрузку стекловаты"
        )
        if file_path:
            try:
                wb.save(file_path)
                messagebox.showinfo("Успех", f"Файл сохранён:\n{file_path}")
            except Exception as e:
                messagebox.showerror("Ошибка", f"Не удалось сохранить: {e}")

class SplashScreen:
    def __init__(self, root, image_path, min_duration=2000):
        self.root = root
        self.min_duration = min_duration
        self.splash_window = None
        self.image_path = image_path
        self.photo = None
        self.progress = None
        self.start_time = None
        self.splash_width = 0
        self.splash_height = 0
        self.app_ready = False  # ← НОВЫЙ ФЛАГ
        self.create_splash()
    
    def create_splash(self):
        self.splash_window = ctk.CTkToplevel(self.root)
        self.splash_window.title("")
        self.splash_window.overrideredirect(True)  # Убираем рамку
        
        # === КЛЮЧЕВОЙ МОМЕНТ: Прозрачность фона ===
        self.splash_window.attributes('-transparentcolor', '#000001')
        self.splash_window.configure(fg_color='#000001')  # Цвет фона окна
        # ==========================================
        
        screen_width = self.splash_window.winfo_screenwidth()
        screen_height = self.splash_window.winfo_screenheight()
        
        try:
            # Открываем изображение с сохранением прозрачности (RGBA)
            self.original_image = Image.open(self.image_path).convert("RGBA")
            orig_width, orig_height = self.original_image.size
            
            # === Масштаб 1/3 ===
            scale_factor = 1/3
            # ==================
            
            # Максимальные размеры (с отступами от краёв экрана)
            max_width = min(orig_width, screen_width - 100)
            max_height = min(orig_height, screen_height - 100)
            
            # Сохраняем соотношение сторон
            ratio = min(max_width / orig_width, max_height / orig_height)
            
            # Применяем масштабирование 1/3
            self.splash_width = int(orig_width * ratio * scale_factor)
            self.splash_height = int(orig_height * ratio * scale_factor)
            
            # Ресайз с сохранением качества и прозрачности
            self.resized_image = self.original_image.resize(
                (self.splash_width, self.splash_height), 
                Image.Resampling.LANCZOS
            )
            
            self.photo = ctk.CTkImage(
                light_image=self.resized_image,
                dark_image=self.resized_image,
                size=(self.splash_width, self.splash_height)
            )
            
            # Label с прозрачным фоном
            label = ctk.CTkLabel(
                self.splash_window, 
                image=self.photo, 
                text="",
                fg_color='#000001'  # Тот же цвет, что и прозрачный фон окна
            )
            label.place(x=0, y=0, relwidth=1.0, relheight=1.0)
            
        except Exception as e:
            print(f"⚠️ Не удалось загрузить изображение заставки: {e}")
            self.splash_width = 500
            self.splash_height = 300
            self.splash_window.geometry(f"{self.splash_width}x{self.splash_height}")
            label = ctk.CTkLabel(
                self.splash_window, 
                text="Metal Accounting App\nЗагрузка...",
                font=ctk.CTkFont(size=24, weight="bold"),
                width=self.splash_width,
                height=self.splash_height
            )
            label.place(x=0, y=0, relwidth=1.0, relheight=1.0)
        
        # Центрирование окна на экране
        x = (screen_width - self.splash_width) // 2
        y = (screen_height - self.splash_height) // 2
        self.splash_window.geometry(f"{self.splash_width}x{self.splash_height}+{x}+{y}")
        self.splash_window.attributes('-topmost', True)
        
        # Прогресс-бар (внизу изображения)
        self.progress = ctk.CTkProgressBar(
            self.splash_window, 
            mode="indeterminate",
            width=self.splash_width - 40,
            height=10,
            fg_color='#000001'  # Прозрачный фон прогресс-бара
        )
        self.progress.place(x=20, y=self.splash_height - 30)
        self.progress.start()
        
        self.start_time = time.time()
        self.root.update()
    
    def can_close(self):
        """Проверяет, можно ли закрыть заставку"""
        if self.start_time is None:
            return self.app_ready
        elapsed = (time.time() - self.start_time) * 1000
        # ← ИСПРАВЛЕНИЕ: ждём И время, И готовность приложения
        return elapsed >= self.min_duration and self.app_ready
    
    def set_app_ready(self):
        """Вызывается когда приложение полностью загружено"""
        self.app_ready = True
    
    def close(self):
        while not self.can_close():
            time.sleep(0.1)
            self.root.update()
        if self.progress:
            self.progress.stop()
        if self.splash_window:
            self.splash_window.destroy()
            self.splash_window = None

def main():
    root = ctk.CTk()
    root.withdraw()
    
    # Путь к изображению заставки
    splash_image_path = r"\\f12\exchange\Production\BD METALL do not delete\radax.png"
    
    # Создаем заставку (используем КЛАСС ИЗ НАЧАЛА ФАЙЛА)
    splash = SplashScreen(root, splash_image_path, min_duration=2000)
    
    app_instance = [None]
    db_connection = [None]  # ← Передаем БД через список
    
    def load_app_in_background():
        """Загрузка приложения в фоновом потоке (БЕЗ работы с SQLite!)"""
        try:
            # Создаем приложение БЕЗ синхронизации с БД в этом потоке
            app = MetalAccountingApp(root)
            app_instance[0] = app
            # Не трогаем БД в этом потоке!
            splash.set_app_ready()  # Сигнал заставке
        except Exception as e:
            print(f"❌ Ошибка загрузки приложения: {e}")
            import traceback
            traceback.print_exc()
            splash.app_ready = True
    
    def check_and_show():
        """Проверяет готовность и показывает приложение"""
        if splash.can_close():
            splash.close()
            root.deiconify()
            if app_instance[0]:
                app_instance[0].status_var.set("Готов")
                # ← БД синхронизируем ТОЛЬКО в главном потоке!
                if app_instance[0].db and app_instance[0].db.connection:
                    app_instance[0].start_auto_sync()
        else:
            root.after(100, check_and_show)
    
    # Запускаем загрузку в отдельном потоке
    load_thread = threading.Thread(target=load_app_in_background, daemon=True)
    load_thread.start()
    
    # Начинаем проверку готовности
    root.after(100, check_and_show)
    
    root.mainloop()


if __name__ == "__main__":
    main()
